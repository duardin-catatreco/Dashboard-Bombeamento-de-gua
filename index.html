create or replace function public.resumo_24h(p_device text) returns table ( medido_em timestamptz, distancia_cm numeric, vazao_max numeric, bomba_alguma boolean ) language sql stable security invoker set search_path = public as $$ select distinct on (date_trunc('hour', m.medido_em)) m.medido_em, m.distancia_cm, max(m.vazao_lpm) over w, bool_or(m.bomba_ligada) over w from public.medicoes m where m.device_id = p_device and m.medido_em >= now() - interval '24 hours' and m.distancia_cm is not null window w as (partition by date_trunc('hour', m.medido_em)) order by date_trunc('hour', m.medido_em), m.medido_em desc; $$; grant execute on function public.resumo_24h(text) to anon, authenticated;
