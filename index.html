-- ===================================================================== -- 08_resumo_dashboard.sql · Resumo de 24 h para o gráfico do dashboard -- Execute UMA vez no SQL Editor (pode repetir sem problema). -- Devolve 1 linha por hora (a última leitura de cada hora, com a maior -- vazão e se a bomba ligou em algum momento naquela hora). Assim o site -- baixa ~24 linhas em vez de dezenas de milhares. -- Respeita a RLS: só enxerga o que as políticas de leitura pública deixam. -- ===================================================================== create or replace function public.resumo_24h(p_device text) returns table ( medido_em timestamptz, distancia_cm numeric, vazao_max numeric, bomba_alguma boolean ) language sql stable security invoker set search_path = public as $$ select distinct on (date_trunc('hour', m.medido_em)) m.medido_em, m.distancia_cm, max(m.vazao_lpm) over w, bool_or(m.bomba_ligada) over w from public.medicoes m where m.device_id = p_device and m.medido_em >= now() - interval '24 hours' and m.distancia_cm is not null window w as (partition by date_trunc('hour', m.medido_em)) order by date_trunc('hour', m.medido_em), m.medido_em desc; $$; grant execute on function public.resumo_24h(text) to anon, authenticated; -- Teste (deve devolver algumas linhas se o ESP32 já enviou dados): -- select * from public.resumo_24h('esp32_bombeamento_01');
